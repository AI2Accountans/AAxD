  <script>
    // Particle Animation & Canvas Engine
    const canvas = document.getElementById('flowCanvas');
    const ctx = canvas.getContext('2d');
    const container = document.querySelector('.player-container');
    const colLeft = document.getElementById('colLeft');
    const colCenter = document.getElementById('colCenter');
    const colRight = document.getElementById('colRight');

    let isPlaying = true;
    let currentTime = 14.2;
    const totalDuration = 28.0;
    let simulationMode = 'normal'; // 'normal', 'blocked', 'audit'

    function resizeCanvas() {
      const rect = container.getBoundingClientRect();
      canvas.width = rect.width;
      canvas.height = rect.height;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    // Particle class
    class Particle {
      constructor(type) {
        this.reset(type);
      }

      reset(type) {
        this.type = type || simulationMode;
        const leftRect = colLeft.getBoundingClientRect();
        const centerRect = colCenter.getBoundingClientRect();
        const rightRect = colRight.getBoundingClientRect();
        const cRect = container.getBoundingClientRect();

        if (this.type === 'audit') {
          // Flows from Right to Center, then back
          this.startX = rightRect.left - cRect.left + 20;
          this.startY = rightRect.top - cRect.top + 60 + Math.random() * (rightRect.height - 120);
          this.targetX = centerRect.right - cRect.left - 20;
          this.targetY = centerRect.top - cRect.top + 80 + Math.random() * (centerRect.height - 160);
          this.color = '#a855f7';
        } else {
          // Flows from Left to Center
          this.startX = leftRect.right - cRect.left - 10;
          this.startY = leftRect.top - cRect.top + 40 + Math.random() * (leftRect.height - 80);
          this.targetX = centerRect.left - cRect.left + 20;
          this.targetY = centerRect.top - cRect.top + 60 + Math.random() * (centerRect.height - 120);
          this.color = this.type === 'blocked' ? '#f43f5e' : '#38bdf8';
        }

        this.x = this.startX;
        this.y = this.startY;
        this.progress = 0;
        this.speed = 0.008 + Math.random() * 0.008;
        this.size = 3.5 + Math.random() * 2;
        this.tail = [];
        this.phase = 1; // 1: left->center, 2: center->right
      }

      update() {
        if (!isPlaying) return;

        this.tail.push({ x: this.x, y: this.y });
        if (this.tail.length > 8) this.tail.shift();

        this.progress += this.speed;

        if (this.phase === 1) {
          // Quadratic bezier curve to center
          const midX = (this.startX + this.targetX) / 2;
          const cpY = this.startY + (this.targetY - this.startY) * 0.1;
          const t = this.progress;
          this.x = (1 - t) * (1 - t) * this.startX + 2 * (1 - t) * t * midX + t * t * this.targetX;
          this.y = (1 - t) * (1 - t) * this.startY + 2 * (1 - t) * t * cpY + t * t * this.targetY;

          if (this.progress >= 1) {
            if (this.type === 'blocked') {
              // Particle is rejected at Layer 01 SHACL!
              flashLayer('layer01', '#f43f5e');
              this.reset('blocked');
            } else if (this.type === 'audit') {
              flashLayer('layer03', '#a855f7');
              this.reset('audit');
            } else {
              // Valid! Advances through Virtuoso to Right Column
              flashLayer('layer02', '#38bdf8');
              this.phase = 2;
              this.progress = 0;
              const centerRect = colCenter.getBoundingClientRect();
              const rightRect = colRight.getBoundingClientRect();
              const cRect = container.getBoundingClientRect();

              this.startX = this.x;
              this.startY = this.y;
              this.targetX = rightRect.left - cRect.left + 20;
              this.targetY = rightRect.top - cRect.top + 40 + Math.random() * (rightRect.height - 80);
              this.color = '#10b981';
            }
          }
        } else if (this.phase === 2) {
          const midX = (this.startX + this.targetX) / 2;
          const t = this.progress;
          this.x = (1 - t) * (1 - t) * this.startX + 2 * (1 - t) * t * midX + t * t * this.targetX;
          this.y = (1 - t) * this.startY + t * this.targetY;

          if (this.progress >= 1) {
            this.reset();
          }
        }
      }

      draw() {
        // Draw glow trail
        ctx.beginPath();
        for (let i = 0; i < this.tail.length; i++) {
          const pt = this.tail[i];
          const alpha = (i / this.tail.length) * 0.4;
          ctx.fillStyle = this.color;
          ctx.globalAlpha = alpha;
          ctx.fillRect(pt.x - 1.5, pt.y - 1.5, 3, 3);
        }
        ctx.globalAlpha = 1;

        // Draw particle head
        ctx.shadowBlur = 10;
        ctx.shadowColor = this.color;
        ctx.fillStyle = '#ffffff';
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size / 2, 0, Math.PI * 2);
        ctx.fill();

        ctx.strokeStyle = this.color;
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.stroke();
        ctx.shadowBlur = 0;
      }
    }

    // Flash animation on stack layers
    function flashLayer(layerId, color) {
      const el = document.getElementById(layerId);
      if (!el) return;
      el.style.borderColor = color;
      el.style.boxShadow = `0 0 25px ${color}`;
      setTimeout(() => {
        el.style.borderColor = '';
        el.style.boxShadow = '';
      }, 350);
    }

    const particles = [];
    for (let i = 0; i < 22; i++) {
      particles.push(new Particle());
    }

    function animate() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      particles.forEach(p => {
        p.update();
        p.draw();
      });

      // Update timer & scrubber
      if (isPlaying) {
        currentTime += 0.016;
        if (currentTime >= totalDuration) {
          currentTime = 0;
        }
        updateScrubberUI();
      }

      requestAnimationFrame(animate);
    }
    requestAnimationFrame(animate);

    // Scrubber & Play UI
    const playPauseBtn = document.getElementById('playPauseBtn');
    const playIcon = document.getElementById('playIcon');
    const pauseIcon = document.getElementById('pauseIcon');
    const timeCounter = document.getElementById('timeCounter');
    const scrubberProgress = document.getElementById('scrubberProgress');
    const scrubberThumb = document.getElementById('scrubberThumb');
    const scrubberTrack = document.getElementById('scrubberTrack');

    function updateScrubberUI() {
      const pct = (currentTime / totalDuration) * 100;
      scrubberProgress.style.width = pct + '%';
      scrubberThumb.style.left = pct + '%';

      const sec = currentTime.toFixed(1);
      timeCounter.textContent = `0:${sec.padStart(4, '0')} / 0:${totalDuration.toFixed(1)}`;
    }

    playPauseBtn.addEventListener('click', () => {
      isPlaying = !isPlaying;
      playIcon.style.display = isPlaying ? 'none' : 'block';
      pauseIcon.style.display = isPlaying ? 'block' : 'none';
    });

    scrubberTrack.addEventListener('click', (e) => {
      const rect = scrubberTrack.getBoundingClientRect();
      const clickX = e.clientX - rect.left;
      const pct = Math.max(0, Math.min(1, clickX / rect.width));
      currentTime = pct * totalDuration;
      updateScrubberUI();
    });

    // Fullscreen Toggle
    document.getElementById('fullscreenBtn').addEventListener('click', () => {
      if (!document.fullscreenElement) {
        container.requestFullscreen?.() || container.webkitRequestFullscreen?.();
      } else {
        document.exitFullscreen?.() || document.webkitExitFullscreen?.();
      }
    });

    // Mode Selector Buttons
    const btnModeNormal = document.getElementById('btnModeNormal');
    const btnModeBlocked = document.getElementById('btnModeBlocked');
    const btnModeAudit = document.getElementById('btnModeAudit');

    function setMode(mode, activeBtn) {
      simulationMode = mode;
      document.querySelectorAll('.mode-btn').forEach(b => b.classList.remove('active'));
      activeBtn.classList.add('active');
      particles.forEach(p => p.reset(mode));
    }

    btnModeNormal.addEventListener('click', () => setMode('normal', btnModeNormal));
    btnModeBlocked.addEventListener('click', () => setMode('blocked', btnModeBlocked));
    btnModeAudit.addEventListener('click', () => setMode('audit', btnModeAudit));
  </script>
